const fs = require("fs"); const http = require("http"); const server = http.createServer((req, res) => { fs.readFile("message.txt", "utf8", (err, data) => { if (err) { res.writeHead(500, { "Content-Type": "text/html" }); res.end("

Erreur serveur

"); } else { res.writeHead(200, { "Content-Type": "text/html" }); res.end(`

Mon serveur Node.js

${data}

© Rawen
`); } }); }); server.listen(8080, () => console.log("Serveur sur le port 8080"));
